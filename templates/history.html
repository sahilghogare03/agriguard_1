To persist history across sessions to Supabase, add your credentials in the .env file:

SUPABASE_URL=https://your-project-id.supabase.co

SUPABASE_KEY=your-anon-or-service-key

Supabase SQL Editor Table & Permission Setup Query:

-- 1. Create table (if not created yet)
CREATE TABLE IF NOT EXISTS prediction_history (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    disease TEXT NOT NULL,
    confidence FLOAT NOT NULL,
    health_status TEXT NOT NULL,
    description TEXT,
    treatment TEXT,
    prevention TEXT,
    image_path TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT TIMEZONE('utc'::text, NOW()) NOT NULL
);

-- 2. Grant access & disable RLS to allow reading and inserting
ALTER TABLE prediction_history DISABLE ROW LEVEL SECURITY;
GRANT ALL ON TABLE prediction_history TO anon, authenticated, service_role;
